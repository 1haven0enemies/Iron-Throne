# For the Duel Leader Title, Leader Names, and custom portrait!

defined_text = {
	name = GetFirstTitle
	text = {
		trigger = {
			has_country_flag = unique_first_title
		}
		localization_key = "[THIS.GetTag]_FIRST_LEADER_TITLE"
	}
	text = {
		localization_key = DEFAULT_FIRST_TITLE
	}
}

defined_text = {
	name = GetSecondTitle
	text = {
		trigger = {
			check_variable = {THIS.currentdualleader > 0 }
		}
		localization_key = "[THIS.GetTag]_SECOND_LEADER_TITLE"
	}
}

defined_text = {
	name = GetDualLeaderName
	text = {
		trigger = {
			check_variable = {THIS.currentdualleader > 0 }
		}
		localization_key = "[THIS.GetTag]_SECOND_LEADER_NAME_[?currentdualleader]"
	}
	text = {
		localization_key = DEFAULT_TITLE
	}
}

defined_text = {
    name = GetDualLeaderPortrait
	text = {
		trigger = {
			check_variable = { THIS.currentdualleader > 0 }
		}
		localization_key = "GFX_[THIS.GetTag]_dual_leader_[?currentdualleader]"
	}
	text = {
		trigger = {
			check_variable = { currentdualleader = 0 }
		}
		localization_key = "GFX_portrait_bg_blank"
	}
}

defined_text = {
    name = GetDualLeaderDesc

        }

	}
}
